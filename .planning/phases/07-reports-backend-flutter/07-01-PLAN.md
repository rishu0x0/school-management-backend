```go
package reports

import "time"

// ReportData is the input to both PDF and Excel generators.
type ReportData struct {
	ClassName  string
	Month      string // YYYY-MM
	GeneratedAt time.Time
	// Days is the sorted list of calendar days that had at least one record
	// OR all days in the month — caller decides. Max 31.
	Days []int // 1-31
	// Students ordered by roll number.
	Students []StudentRow
}

// StudentRow holds one student's attendance for the report month.
type StudentRow struct {
	Name       string
	RollNumber int
	// Records maps day-of-month → status ("P", "A", "L", "—")
	Records    map[int]string
	// Totals
	Present int
	Absent  int
	Leave   int
	Total   int // days with any record
}

// AttendanceLabel converts a DB status string to the display label.
func AttendanceLabel(status string) string {
	switch status {
	case "present":
		return "P"
	case "absent":
		return "A"
	case "leave":
		return "L"
	default:
		return "—"
	}
}
```

```go
package reports

import (
	"bytes"
	"fmt"

	"github.com/johnfercher/maroto/v2"
	"github.com/johnfercher/maroto/v2/pkg/components/col"
	"github.com/johnfercher/maroto/v2/pkg/components/row"
	"github.com/johnfercher/maroto/v2/pkg/components/text"
	"github.com/johnfercher/maroto/v2/pkg/config"
	"github.com/johnfercher/maroto/v2/pkg/consts/align"
	"github.com/johnfercher/maroto/v2/pkg/consts/fontstyle"
	"github.com/johnfercher/maroto/v2/pkg/consts/orientation"
	"github.com/johnfercher/maroto/v2/pkg/consts/pagesize"
	"github.com/johnfercher/maroto/v2/pkg/props"
)

const studentsPerPage = 25

// GeneratePDF generates a PDF attendance report and returns raw bytes.
func GeneratePDF(data ReportData) ([]byte, error) {
	cfg := config.NewBuilder().
		WithPageSize(pagesize.A4).
		WithOrientation(orientation.Horizontal).
		WithLeftMargin(10).
		WithRightMargin(10).
		WithTopMargin(10).
		WithBottomMargin(10).
		Build()

	mrt := maroto.New(cfg)

	addCoverPage(mrt, data)
	addDataPages(mrt, data)
	addSummaryPage(mrt, data)

	doc, err := mrt.Generate()
	if err != nil {
		return nil, fmt.Errorf("pdf generate: %w", err)
	}

	var buf bytes.Buffer
	if err := doc.Save(&buf); err != nil {
		return nil, fmt.Errorf("pdf save: %w", err)
	}
	return buf.Bytes(), nil
}

func addCoverPage(m maroto.Maroto, data ReportData) {
	m.AddRow(30,
		col.New(12).Add(text.New(data.ClassName,
			props.Text{Size: 24, Style: fontstyle.Bold, Align: align.Center, Top: 8})),
	)
	m.AddRow(10,
		col.New(12).Add(text.New("Attendance Report — "+data.Month,
			props.Text{Size: 14, Align: align.Center})),
	)
	m.AddRow(8,
		col.New(12).Add(text.New("Generated: "+data.GeneratedAt.Format("02 Jan 2006 15:04 MST"),
			props.Text{Size: 10, Align: align.Center, Color: &props.Color{Red: 100, Green: 100, Blue: 100}})),
	)
	m.AddRow(5,
		col.New(12).Add(text.New(
			fmt.Sprintf("Total Students: %d  |  Days in Report: %d", len(data.Students), len(data.Days)),
			props.Text{Size: 10, Align: align.Center})),
	)
}

func addDataPages(m maroto.Maroto, data ReportData) {
	// Split students into pages of studentsPerPage
	for pageStart := 0; pageStart < len(data.Students); pageStart += studentsPerPage {
		pageEnd := pageStart + studentsPerPage
		if pageEnd > len(data.Students) {
			pageEnd = len(data.Students)
		}
		pageStudents := data.Students[pageStart:pageEnd]

		m.AddRow(5) // spacer / page break marker — maroto auto-breaks

		// Header row
		headerCols := buildHeaderCols(data.Days)
		m.AddRow(7, headerCols...)

		// Student rows
		for _, stu := range pageStudents {
			rowCols := buildStudentCols(stu, data.Days)
			m.AddRow(6, rowCols...)
		}
	}
}

func buildHeaderCols(days []int) []core.Col {
	// 12-grid: name=3, roll=1, each day≈(12-4)/len(days) but we cap days display
	cols := []core.Col{
		col.New(3).Add(text.New("Student Name",
			props.Text{Size: 8, Style: fontstyle.Bold})),
		col.New(1).Add(text.New("Roll",
			props.Text{Size: 8, Style: fontstyle.Bold, Align: align.Center})),
	}
	remaining := 8
	dayWidth := 1
	if len(days) > 0 && remaining/len(days) >= 1 {
		dayWidth = remaining / len(days)
		if dayWidth < 1 {
			dayWidth = 1
		}
	}
	for i, d := range days {
		w := dayWidth
		// distribute leftover cols to first columns
		if i < remaining-(dayWidth*len(days)) {
			w++
		}
		if w < 1 {
			w = 1
		}
		cols = append(cols, col.New(w).Add(text.New(
			fmt.Sprintf("%d", d),
			props.Text{Size: 7, Style: fontstyle.Bold, Align: align.Center},
		)))
	}
	return cols
}

func buildStudentCols(stu StudentRow, days []int) []core.Col {
	cols := []core.Col{
		col.New(3).Add(text.New(stu.Name, props.Text{Size: 7})),
		col.New(1).Add(text.New(fmt.Sprintf("%d", stu.RollNumber),
			props.Text{Size: 7, Align: align.Center})),
	}
	remaining := 8
	dayWidth := 1
	if len(days) > 0 && remaining/len(days) >= 1 {
		dayWidth = remaining / len(days)
		if dayWidth < 1 {
			dayWidth = 1
		}
	}
	for i, d := range days {
		w := dayWidth
		if i < remaining-(dayWidth*len(days)) {
			w++
		}
		if w < 1 {
			w = 1
		}
		label := stu.Records[d]
		if label == "" {
			label = "—"
		}
		cellColor := &props.Color{Red: 255, Green: 255, Blue: 255}
		switch label {
		case "P":
			cellColor = &props.Color{Red: 200, Green: 240, Blue: 200}
		case "A":
			cellColor = &props.Color{Red: 255, Green: 200, Blue: 200}
		case "L":
			cellColor = &props.Color{Red: 255, Green: 240, Blue: 180}
		}
		_ = cellColor // maroto v2 text doesn't support cell background; just render text
		cols = append(cols, col.New(w).Add(text.New(label,
			props.Text{Size: 7, Align: align.Center},
		)))
	}
	return cols
}

func addSummaryPage(m maroto.Maroto, data ReportData) {
	m.AddRow(5) // spacer

	m.AddRow(8,
		col.New(12).Add(text.New("Summary",
			props.Text{Size: 12, Style: fontstyle.Bold})),
	)

	// Header
	m.AddRow(7,
		col.New(4).Add(text.New("Student Name", props.Text{Size: 8, Style: fontstyle.Bold})),
		col.New(1).Add(text.New("Roll", props.Text{Size: 8, Style: fontstyle.Bold, Align: align.Center})),
		col.New(1).Add(text.New("P", props.Text{Size: 8, Style: fontstyle.Bold, Align: align.Center})),
		col.New(1).Add(text.New("A", props.Text{Size: 8, Style: fontstyle.Bold, Align: align.Center})),
		col.New(1).Add(text.New("L", props.Text{Size: 8, Style: fontstyle.Bold, Align: align.Center})),
		col.New(2).Add(text.New("Days", props.Text{Size: 8, Style: fontstyle.Bold, Align: align.Center})),
		col.New(2).Add(text.New("%", props.Text{Size: 8, Style: fontstyle.Bold, Align: align.Center})),
	)

	for _, stu := range data.Students {
		pct := 0.0
		if stu.Total > 0 {
			pct = float64(stu.Present) / float64(stu.Total) * 100
		}
		m.AddRow(6,
			col.New(4).Add(text.New(stu.Name, props.Text{Size: 7})),
			col.New(1).Add(text.New(fmt.Sprintf("%d", stu.RollNumber), props.Text{Size: 7, Align: align.Center})),
			col.New(1).Add(text.New(fmt.Sprintf("%d", stu.Present), props.Text{Size: 7, Align: align.Center})),
			col.New(1).Add(text.New(fmt.Sprintf("%d", stu.Absent), props.Text{Size: 7, Align: align.Center})),
			col.New(1).Add(text.New(fmt.Sprintf("%d", stu.Leave), props.Text{Size: 7, Align: align.Center})),
			col.New(2).Add(text.New(fmt.Sprintf("%d", stu.Total), props.Text{Size: 7, Align: align.Center})),
			col.New(2).Add(text.New(fmt.Sprintf("%.1f%%", pct), props.Text{Size: 7, Align: align.Center})),
		)
	}
}
```
